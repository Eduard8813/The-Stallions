@startuml
' Wani Connect (The Stallions) - Diagrama de clases
' Coherente con el modelo ER (mismas entidades y multiplicidades)
'
' Relaciones:
'   -- asociacion | o-- agregacion | *-- composicion | <|-- herencia
hide circle
skinparam classAttributeIconSize 0

enum AuthProvider {
  LOCAL
  GOOGLE
  FIREBASE
}

enum Visibilidad {
  PUBLICO
  PRIVADO
}

enum CategoriaEvento {
  FERIA
  TALLER
  CONFERENCIA
  CULTURAL
  OTRO
}

enum Rol {
  USER
  ADMIN
  AUDITOR
}

class User {
  - id: Long
  - email: String
  - passwordHash: String
  - fullName: String
  - rol: Rol
  - provider: AuthProvider
  - providerId: String
  - photoUrl: String
  - photoData: byte[]
  - photoContentType: String
  - username: String
  - phone: String
  - birthDate: LocalDate
  - gender: String
  - city: String
  - bio: String
  - twoFactorEnabled: boolean
  - totpSecret: String
  - enabled: boolean
  - createdAt: LocalDateTime
  - updatedAt: LocalDateTime
  + actualizarPerfil(datos): void
  + cambiarFoto(foto): void
  + desactivarCuenta(): void
  + activar2FA(secret: String): void
  + desactivar2FA(): void
}

class Photo {
  - id: Long
  - visibilidad: Visibilidad
  - url: String
  - photoData: byte[]
  - contentType: String
  - descripcion: String
  - fechaUpload: LocalDateTime
  + subir(foto): void
  + editarDescripcion(texto): void
}

class PhotoGroup {
  - id: Long
  - createdAt: LocalDateTime
  + crearGrupo(): PhotoGroup
}

class PhotoLike {
  - user: User
  - photo: Photo
  - createdAt: LocalDateTime
  + registrar(user, photo): void
  + retirar(): void
}

class Comment {
  - id: Long
  - texto: String
  - fecha: LocalDateTime
  - editado: boolean
  + agregar(texto): void
  + editar(texto): void
}

class Notification {
  - id: Long
  - tipo: String
  - mensaje: String
  - relacionId: String
  - leida: boolean
  - fecha: LocalDateTime
  + marcarComoLeida(): void
  + notificar(origen, tipo): void
}

class NotificationChannelPreference {
  - user: User
  - categoria: String
  - canal: String
  - habilitado: boolean
  + actualizarPreferencia(canal, habilitado): void
}

class UserPrivacy {
  - user: User
  - visibility: String
  - showEmail: boolean
  - showPhone: boolean
  - showLocation: boolean
  - discoverable: boolean
  + actualizar(config): void
}

class BlockedUser {
  - id: Long
  - blockedUser: User
  - name: String
  - username: String
  - createdAt: LocalDateTime
  + bloquear(usuario): void
  + desbloquear(): void
}

class DataExport {
  - id: Long
  - exportId: String
  - status: String
  - availableForHours: int
  - createdAt: LocalDateTime
  + solicitarExportacion(user): void
  + eliminar(): void
}

class UserSession {
  - id: Long
  - device: String
  - platform: String
  - location: String
  - lastActive: LocalDateTime
  - active: boolean
  - tokenId: String
  + revocar(): void
  + registrarActividad(): void
  + estaExpirada(minutosInactividad: int): boolean
}

class RoleChangeAudit {
  - id: Long
  - changedById: Long
  - changedByEmail: String
  - fromRole: Rol
  - toRole: Rol
  - createdAt: LocalDateTime
  + registrarCambio(usuario, admin, anterior, nuevo): void
}

class TwoFactorChallenge {
  - id: Long
  - challengeId: String
  - codeHash: String
  - expiresAt: LocalDateTime
  - used: boolean
  + verificarCodigo(codigo: String): boolean
}

class Emprendimiento {
  - id: Long
  - nombre: String
  - tipo: String
  - descripcion: String
  - lat: Double
  - lng: Double
  - contactoTelefono: String
  - contactoEmail: String
  - contactoRedes: String
  + publicar(): void
  + actualizar(datos): void
}

class Evento {
  - id: Long
  - titulo: String
  - fecha: LocalDate
  - fechaFin: LocalDate
  - categoria: CategoriaEvento
  - descripcion: String
  + publicar(): void
  + tieneRango(): boolean
}

' Composicion: el aggreado de los usuarios aguanta sus fotos, comentarios y sesiones
User *-- PhotoGroup : "1" o-- "*"
User *-- "0..*" Comment : escribe
User *-- "0..*" UserSession
User *-- "0..1" UserPrivacy
User *-- "0..*" NotificationChannelPreference
User "1" o-- "0..*" RoleChangeAudit : registra

' Asociaciones
Photo "1" *-- "0..*" Comment
Photo "0..*" -- "1" User : publica
PhotoGroup "1" o-- "0..*" Photo : agrupa fotos
PhotoLike "0..*" -- "1" User
PhotoLike "0..*" -- "1" Photo

Notification "0..*" -- "1" User : destino
Notification "0..*" -- "0..1" User : origen
BlockedUser "0..*" -- "1" User : bloquea
BlockedUser "0..*" -- "1" User : bloqueado
DataExport "0..*" -- "1" User
TwoFactorChallenge "0..*" -- "1" User
Emprendimiento "0..*" -- "1" User : crea
Evento "0..*" -- "1" User : crea

@enduml
